{{- /* Previous / next part of the same series, for a part page (classes/single.html). Context: dict "series" "seriesSlug" "partIndex". Only section items have part pages (the content adapter), so lectures directly under a series are skipped. */ -}} {{- $prev := false }} {{- $next := false }} {{- range $i, $item := .series.items }} {{- if eq $item.type "section" }} {{- if lt $i $.partIndex }}{{ $prev = $item }}{{ end }} {{- if and (gt $i $.partIndex) (not $next) }}{{ $next = $item }}{{ end }} {{- end }} {{- end }} {{- if or $prev $next }}
{{- with $prev }} ← Previous {{ .title }} {{- end }} {{- with $next }} Next → {{ .title }} {{- end }}
{{- end }}
