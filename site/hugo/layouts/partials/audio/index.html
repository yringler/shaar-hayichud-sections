{{- /* What audio data the data build published (data/dist/v1, mounted as assets/data/v1). Call with partialCached "audio/index.html" . "audio". Returns dict "chapters": chapter key -> asset path of its chapter alignment index "aligned": lecture id -> dict "chapter" (key) "url" (English chapter page) "lines" (aligned lines there) "all" (the same dicts for every chapter the class reads, in chapter order). "chapter"/"url" are the chapter with the most aligned lines (a class that ends one chapter and starts the next is listed on both). "outlines": lecture id -> asset path of its topic outline (Transcripts are not published: see data/README.md.) */ -}} {{- $cIndex := resources.Get "data/v1/alignments/chapters/index.json" -}} {{- if not $cIndex -}} {{- errorf "audio indexes missing under data/dist/v1: run the data build first (yarn build / yarn data in site/)" -}} {{- end -}} {{- $s := newScratch -}} {{- $s.Set "chapters" dict -}} {{- $s.Set "aligned" dict -}} {{- $s.Set "alignedAll" dict -}} {{- $s.Set "outlines" dict -}} {{- with resources.Get "data/v1/outlines/index.json" -}} {{- range (. | transform.Unmarshal).items -}} {{- $s.SetInMap "outlines" .id (printf "data/v1/outlines/%s.json" .id) -}} {{- end -}} {{- end -}} {{- range ($cIndex | transform.Unmarshal).items -}} {{- $key := .id -}} {{- $path := printf "data/v1/%s" .path -}} {{- $s.SetInMap "chapters" $key $path -}} {{- $chapterUrl := printf "/texts/chapter-%s/" (replaceRE `^chapter_0*` "" $key) -}} {{- $data := resources.Get $path | transform.Unmarshal -}} {{- $counts := newScratch -}} {{- range $data.lines -}}{{- range .refs -}}{{- $counts.Add (string .lecture) 1 -}}{{- end -}}{{- end -}} {{- range $i, $l := $data.lectures -}} {{- $entry := dict "chapter" $key "url" $chapterUrl "lines" (($counts.Get (string $i)) | default 0) -}} {{- $prev := (index ($s.Get "alignedAll") $l.id) | default slice -}} {{- $s.SetInMap "alignedAll" $l.id ($prev | append (slice $entry)) -}} {{- end -}} {{- end -}} {{- range $id, $list := $s.Get "alignedAll" -}} {{- $best := index $list 0 -}} {{- range $list -}}{{- if gt .lines $best.lines -}}{{- $best = . -}}{{- end -}}{{- end -}} {{- $s.SetInMap "aligned" $id (merge $best (dict "all" $list)) -}} {{- end -}} {{- return dict "chapters" ($s.Get "chapters") "aligned" ($s.Get "aligned") "outlines" ($s.Get "outlines") -}}
