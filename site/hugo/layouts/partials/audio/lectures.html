{{- /* Every catalog lecture by id, with its place in the catalog: dict "lecture" (catalog lecture) "series" (series section) "seriesUrl" "part" (the series' child section it is under, or nil) "partUrl" (its class page: the part page, or the series page for a lecture directly under the series) "anchor" (the id of its box there, class-) "classUrl" (partUrl#anchor) "headings" (titles between the series and the lecture). URLs come from classes-url.html. Call with partialCached "audio/lectures.html" . "lectures". */ -}} {{- $catalog := partial "catalog.html" . -}} {{- /* Scratch, not merge: merge would lowercase the (case-sensitive) ids. */ -}} {{- $out := newScratch -}} {{- range $series := $catalog.series -}} {{- $seriesUrl := partial "classes-url.html" (dict "series" $series) -}} {{- range $a := $series.items -}} {{- $partUrl := $seriesUrl -}} {{- $found := slice -}} {{- if eq $a.type "lecture" -}} {{- $found = slice (dict "lecture" $a "headings" slice) -}} {{- else -}} {{- $partUrl = partial "classes-url.html" (dict "series" $series "part" $a) -}} {{- range $b := $a.items -}} {{- if eq $b.type "lecture" -}} {{- $found = $found | append (slice (dict "lecture" $b "headings" (slice $a.title))) -}} {{- else -}} {{- range $c := $b.items -}} {{- if eq $c.type "lecture" -}} {{- $found = $found | append (slice (dict "lecture" $c "headings" (slice $a.title $b.title))) -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- range $found -}} {{- $anchor := printf "class-%s" (.lecture.id | urlize) -}} {{- $out.SetInMap "m" .lecture.id (dict "lecture" .lecture "series" $series "seriesUrl" $seriesUrl "part" (cond (eq $a.type "lecture") nil $a) "partUrl" $partUrl "headings" .headings "anchor" $anchor "classUrl" (printf "%s#%s" $partUrl $anchor)) -}} {{- end -}} {{- end -}} {{- end -}} {{- return ($out.Get "m") | default dict -}}
